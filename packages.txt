# What Deep Candle Wayland needs: the window manager, the bar, the launcher, the wallpaper, and
# what their scripts run. No apps. Arch and Artix use the same names. rhine is not packaged:
# README, "Install".
#   sudo pacman -S --needed $(awk '/^\[/{s=$0;next} /^[a-z]/ && s=="[pacman]"' packages.txt)
# [optional] is NOT installed by that line: add a piece only if you want what it gives.
# One name per line; a comment on its own line.

[pacman]
# the window manager: river (the compositor; it brings Xwayland), git and zig to build rhine,
# D-Bus for the session (dbus-run-session), polkit for the seat
river
git
zig
dbus
polkit
# the bar, the launcher and its menus, the wallpaper
waybar
fuzzel
swaybg
# the planets' glyphs on the bar (the bar's own fonts are in fonts/)
ttf-dejavu
# the bar's sound and music pieces, and the volume and media keys (pactl, playerctl)
pipewire
pipewire-pulse
wireplumber
libpulse
playerctl
# the bar's weather (curl); window screenshots (jq)
curl
jq
# the screenshot and colour-picker keys
grim
slurp
wl-clipboard

[optional]
# the bar's network piece and its Wi-Fi menu (nmcli). Without it that piece says offline.
networkmanager
iw
# mpd as a music source for the bar (playerctl covers browsers, mpv and the like)
mpc
# notices for volume, brightness, screenshots (any notification daemon that dunstify can reach)
dunst
# marking a screenshot up (Arch only; Artix: build it, github.com/jtheoof/swappy)
swappy
